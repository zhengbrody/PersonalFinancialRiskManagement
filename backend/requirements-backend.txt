# MindMarket backend (FastAPI) — Phase 1 dependencies.
#
# These are ADDITIONAL packages the backend needs on top of the
# existing root-level ``requirements.txt`` (numpy, pandas, pydantic,
# supabase, anthropic, etc.). Pip-install both in CI / Docker:
#
#     pip install -r requirements.txt
#     pip install -r backend/requirements-backend.txt
#
# Kept deliberately small: every line is justified below.

# FastAPI + the recommended Starlette / pydantic toolchain.
# Pinned to a minor range so a major bump can't silently break us.
fastapi>=0.115,<0.120
uvicorn[standard]>=0.30,<0.35

# Supabase JWT verification. PyJWT handles JWT parsing / claims; the
# cryptography backend is required for Supabase projects that sign
# access tokens with asymmetric ES256 / RS256 keys exposed through
# JWKS.
PyJWT>=2.8,<3
cryptography>=42,<49

# httpx is FastAPI's recommended async HTTP client AND drives the
# TestClient used in backend/tests/. Already pulled in transitively by
# fastapi[all], but we list it so the explicit ``pip install -r`` works
# even if upstream extras change.
httpx>=0.27,<1

# MCP — Anthropic's Model Context Protocol SDK. Powers the standalone
# tool server in backend/mcp_server/ that any Claude-compatible client
# (Claude Desktop, Claude Code) can connect to via stdio. Optional at
# runtime (only the ``python -m backend.mcp_server`` entry point
# imports it), but pinned here so a fresh contributor install gets a
# working MCP environment without an extra step.
mcp>=1.0,<2

# Sentry — error tracking. The FastAPI integration auto-captures unhandled
# 500s (e.g. an uncaught ValidationError) with full context. Only initialised
# in production (see app/main.py:_maybe_init_sentry); a no-op otherwise.
sentry-sdk[fastapi]>=2.0,<3

# lxml — required by yfinance 1.4.x's get_earnings_dates() (it parses the Yahoo
# earnings calendar via pandas.read_html, which needs an HTML backend). Without
# it that call raises ImportError and the free-yfinance EPS beat/miss fallback
# in providers/fmp_provider._yf_earnings silently fail-softs to empty. The rest
# of yfinance (price/.info JSON) does NOT need lxml — only the earnings scrape.
lxml>=5,<7
